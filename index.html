Web Audio API Sound Generator (बिना किसी फाइल के साउंड बजाना) function playSound(type) { try { const audioCtx = new (window.AudioContext || window.webkitAudioContext)(); const osc = audioCtx.createOscillator(); const gainNode = audioCtx.createGain(); osc.connect(gainNode); gainNode.connect(audioCtx.destination); if (type === 'click') { osc.frequency.setValueAtTime(400, audioCtx.currentTime); osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.1); gainNode.gain.setValueAtTime(0.1, audioCtx.currentTime); gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1); osc.start(); osc.stop(audioCtx.currentTime + 0.1); } else if (type === 'magic') { osc.type = 'triangle'; osc.frequency.setValueAtTime(523.25, audioCtx.currentTime); osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.08); osc.frequency.setValueAtTime(783.99, audioCtx.currentTime + 0.16); gainNode.gain.setValueAtTime(0.15, audioCtx.currentTime); gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3); osc.start(); osc.stop(audioCtx.currentTime + 0.3); } } catch (e) { console.log("Audio not supported or blocked"); } } // सभी बटन्स पर ऑटोमेटिक क्लिक साउंड जोड़ना document.addEventListener('click', (e) => { if (e.target.tagName === 'BUTTON') { playSound('click'); } }); // 2. LocalStorage Save & Load System function saveMyDesign() { try { const data = { theme: document.querySelector('.option-btn.active')?.innerText || 'default', rust: isRustActive }; localStorage.setItem('princessStudioSave', JSON.stringify(data)); const status = document.getElementById('saveStatus'); status.innerText = "● Saved Successfully!"; status.style.color = "#34d399"; playSound('magic'); setTimeout(() => { status.innerText = "● Ready"; }, 3000); } catch (e) { alert("Save failed!"); } } function loadMyDesign() { try { const saved = localStorage.getItem('princessStudioSave'); if (saved) { const data = JSON.parse(saved); if (data.rust && !isRustActive) { toggleRustTheme(document.getElementById('rustThemeBtn')); } const status = document.getElementById('saveStatus'); status.innerText = "● Loaded!"; status.style.color = "#60a5fa"; playSound('magic'); setTimeout(() => { status.innerText = "● Ready"; }, 3000); } else { alert("कोई सेव किया हुआ लुक नहीं मिला!"); } } catch (e) { alert("Load failed!"); } } // 5. Advanced Fireworks Animation Engine function triggerFireworks() { playSound('magic'); const canvas = document.getElementById('fireworksCanvas'); canvas.style.display = 'block'; const ctx = canvas.getContext('2d'); canvas.width = window.innerWidth; canvas.height = window.innerHeight; let particles = []; const colors = ['#ff6b81', '#ffc048', '#ff4757', '#2ed573', '#1e90ff', '#b85d30', '#ffd700']; for (let i = 0; i < 120; i++) { particles.push({ x: canvas.width / 2, y: canvas.height / 2, vx: (Math.random() - 0.5) * 12, vy: (Math.random() - 0.5) * 12, radius: Math.random() * 4 + 2, color: colors[Math.floor(Math.random() * colors.length)], alpha: 1, decay: Math.random() * 0.015 + 0.01 }); } function animateFireworks() { ctx.clearRect(0, 0, canvas.width, canvas.height); let activeParticles = 0; particles.forEach(p => { if (p.alpha > 0) { activeParticles++; p.x += p.vx; p.y += p.vy; p.vy += 0.15; // Gravity p.alpha -= p.decay; ctx.save(); ctx.globalAlpha = p.alpha; ctx.beginPath(); ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2); ctx.fillStyle = p.color; ctx.fill(); ctx.restore(); } }); if (activeParticles > 0) { requestAnimationFrame(animateFireworks); } else { canvas.style.display = 'none'; } } animateFireworks(); }
